  \item[\textbf{\ansnum.8}] Example:
\begin{verbatim}
gamma = 0.9
n = 8                          # cells 1..n, cell n is the goal
V = [0.0] * (n + 1)
for sweep in range(50):
    new = V[:]
    for s in range(1, n):
        values = []
        for a in (-1, +1):
            s2 = max(1, s + a)
            r = 10 if s2 == n else -0.5
            values.append(r + gamma * (0 if s2 == n else V[s2]))
        new[s] = max(values)
    V = new
print(V[1:n])
\end{verbatim}
