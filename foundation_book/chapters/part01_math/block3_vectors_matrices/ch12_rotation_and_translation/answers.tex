\begin{enumerate}[label=\textbf{\ansnum.\arabic*}]
  \item $(0,1)$, $(-2,0)$, $(-4,3)$.
  \item $(2\cos45^\circ,2\sin45^\circ)=(\sqrt2,\sqrt2)\approx(1.414,1.414)$.
  \item $R(-90^\circ)=\begin{pmatrix}0&1\\-1&0\end{pmatrix}$, so $(1,1)\mapsto(1,-1)$.
  \item With $c=\cos30^\circ$, $s=\sin30^\circ$: $R^2=\begin{pmatrix}c^2-s^2&-2sc\\2sc&c^2-s^2\end{pmatrix}=\begin{pmatrix}0.5&-0.866\\0.866&0.5\end{pmatrix}=R(60^\circ)$, since $c^2-s^2=0.75-0.25=0.5=\cos60^\circ$ and $2sc=0.866=\sin60^\circ$.
  \item $R^{-1}=R^\top=\begin{pmatrix}0.866&0.5\\-0.5&0.866\end{pmatrix}$. Product: $\begin{pmatrix}c^2+s^2&cs-sc\\sc-cs&s^2+c^2\end{pmatrix}=I$.
  \item $(1,7)$. Matrix $\begin{pmatrix}1&0&-1\\0&1&4\\0&0&1\end{pmatrix}$ times $(2,3,1)^\top$ gives $(1,7,1)^\top$.
  \item Rotation by $180^\circ$: $(x,y)\mapsto(-x,-y)$, giving $(0,0)$, $(-2,0)$, $(0,-1)$. Then adding $(1,1)$: $(1,1)$, $(-1,1)$, $(1,0)$.
  \item $x_W=2+3\cdot0-(-1)\cdot1=3$, $y_W=1+3\cdot1+(-1)\cdot0=4$: the ball is at $(3,4)$. (A north-facing robot: $3$ ahead is $(2,4)$, and $1$ to the right is $(3,4)$.)
  \item $x=2\cos45^\circ+1\cdot\cos90^\circ\approx1.414$, $y=2\sin45^\circ+1\cdot\sin90^\circ\approx1.414+1=2.414$. The tip is at about $(1.414,2.414)$.
  \item Largest distance $l_1+l_2=3$ (arm stretched, $\theta_2=0$). Smallest distance $|l_1-l_2|=1$ (arm folded back, $\theta_2=180^\circ$). For $\theta_2=45^\circ$ the distance is $\sqrt{1.414^2+2.414^2}\approx2.80$, which lies in between.
  \item Example:
\begin{verbatim}
import math

def rot(theta):
    c, s = math.cos(theta), math.sin(theta)
    return [[c, -s], [s, c]]

def apply(M, v):
    return [M[0][0] * v[0] + M[0][1] * v[1],
            M[1][0] * v[0] + M[1][1] * v[1]]

print(apply(rot(math.radians(30)), [3, 1]))   # [2.098..., 2.366...]
\end{verbatim}
\end{enumerate}
