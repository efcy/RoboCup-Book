\begin{enumerate}[label=\textbf{\ansnum.\arabic*}]
  \item $1+4+9+16+25=55$.
  \item $3+5+7+9=24$.
  \item $\displaystyle\sum_{i=1}^5 3i=45$.
  \item $z=6-4+1=3$, with the bias $z=2$.
  \item Mean $=\frac{30}{5}=6$. Variance $=\frac{4+4+0+16+16}{5}=8$.
  \item Errors: $0.2,-0.3,0.4,0.1$. Squares: $0.04+0.09+0.16+0.01=0.30$. MSE $=0.30/4=0.075$.
  \item $\sum(x_i-\bar x)=\sum x_i-n\bar x=n\bar x-n\bar x=0$, because $\sum x_i=n\bar x$ by definition of the mean.
  \item There are $50$ pairs, each adding to $101$: $50\cdot101=5050$. The formula gives $\frac{100\cdot101}{2}=5050$.
  \item Example:
\begin{verbatim}
y    = [1, 0, 1, 1]
yhat = [0.8, 0.3, 0.6, 0.9]
total = 0
for i in range(len(y)):
    total += (y[i] - yhat[i]) ** 2
print(total / len(y))      # 0.075
\end{verbatim}
\end{enumerate}
