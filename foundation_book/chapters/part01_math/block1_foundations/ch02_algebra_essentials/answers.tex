  \item $5x=25$, $x=5$.
  \item $3x-6=2x+4$, so $x=10$.
  \item $a=\dfrac{v-v_0}{t}$ and $b=\hat y-wx$.
  \item $x<-2$ (open circle at $-2$, arrow to the left).
  \item Adding the equations: $3x=9$, $x=3$, then $y=4$. The lines cross at $(3,4)$.
  \item $(x-2)(x-3)=0$, so $x=2$ or $x=3$.
  \item $D=9+16=25>0$, so two solutions: $x=\frac12$ and $x=-2$.
  \item $\frac{2^7}{2^5}=2^2=4$ and $\ln(e^5)=5$.

  \item $x_2=2x_1-3$. At $(3,0)$: $2\cdot3-0-3=3>0$, so $(3,0)$ lies on the positive side (below the line, since the line at $x_1=3$ has height $x_2=3$).
\end{enumerate}
