def match(dets, gts, thr=0.5):
    used, flags = set(), []
    for box, score in sorted(dets, key=lambda d: -d[1]):
        best, best_iou = None, thr
        for k, g in enumerate(gts):
            v = iou(box, g)
            if k not in used and v >= best_iou:
                best, best_iou = k, v
        if best is None:
            flags.append(0)
        else:
            used.add(best); flags.append(1)
    return flags
