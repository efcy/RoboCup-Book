import numpy as np, matplotlib.pyplot as plt
fs = 8000; t = np.arange(0, 1, 1/fs)
x = np.sin(2*np.pi*440*t) + 0.5*np.sin(2*np.pi*1000*t)
x += np.random.default_rng(0).normal(0, 0.2, t.size)
X = np.abs(np.fft.rfft(x*np.hanning(x.size))) * 2 / np.hanning(x.size).sum()
f = np.fft.rfftfreq(x.size, 1/fs)
peaks = f[(X > 0.2) & (X >= np.roll(X, 1)) & (X >= np.roll(X, -1))]
print(peaks)        # about [440. 1000.]
plt.semilogy(f, X); plt.xlabel("Hz"); plt.show()
