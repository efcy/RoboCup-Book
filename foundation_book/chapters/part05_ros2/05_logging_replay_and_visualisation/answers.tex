import numpy as np
def stats(stamps, nominal=1 / 30):
    dt = np.diff(np.sort(stamps))
    print("rate  %.2f Hz" % (1 / dt.mean()))
    print("max gap %.1f ms" % (dt.max() * 1e3))
    print("gaps > 1.5 T:", int((dt > 1.5 * nominal).sum()))
    hist, edges = np.histogram(dt * 1e3, bins=10)
    for h, lo, hi in zip(hist, edges[:-1], edges[1:]):
        print("%5.1f-%5.1f ms %s" % (lo, hi, "#" * (60 * h // hist.max())))
rng = np.random.default_rng(3)
t = np.arange(0, 60, 1 / 30) + rng.normal(0, 0.002, 1800)
stats(t[rng.random(1800) > 0.03])
\end{verbatim}
  The reading from the bag replaces the last three lines by the loop of the chapter that fills \verb!stamps!. The histogram shows a high peak around $33$\,ms and small peaks at multiples of it ($67$, $100$\,ms) for one or two lost frames.
