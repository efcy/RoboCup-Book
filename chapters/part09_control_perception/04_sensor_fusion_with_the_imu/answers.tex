import random, math
random.seed(1)
N, sigma = 400, 0.01
samples = [0.02 + random.gauss(0, sigma) for _ in range(N)]
bias = sum(samples) / N
print(round(bias, 4), "expected std", sigma / math.sqrt(N))

\begin{verbatim}
const G: f64 = 9.81;

struct Comp {
    angle: f64,
    tau: f64,
}

impl Comp {
    fn update(&mut self, gyro: f64, acc: [f64; 3], dt: f64,
              gate: bool) {
        let alpha = self.tau / (self.tau + dt);
        let norm = acc.iter().map(|a| a * a).sum::<f64>().sqrt();
        let trusted = !gate || (norm - G).abs() < 0.5;
        let predicted = self.angle + gyro * dt;
        // without a trusted accelerometer the gyroscope alone decides
        let acc_angle =
            if trusted { acc[0].atan2(acc[2]) } else { predicted };
        self.angle = alpha * predicted + (1.0 - alpha) * acc_angle;
    }
}

fn run(gate: bool) -> f64 {
    let (dt, tilt, bias) = (0.005, 0.10, 0.0);
    let mut f = Comp { angle: tilt, tau: 0.245 };
    let mut worst = 0.0_f64;
    for k in 0..1000 {
        let t = k as f64 * dt;
        let push = if (2.0..2.5).contains(&t) { 8.0 } else { 0.0 };
        let acc = [G * tilt.sin() + push, 0.0, G * tilt.cos()];
        f.update(bias, acc, dt, gate);
        worst = worst.max((f.angle - tilt).abs());
    }
    worst
}

fn main() {
    println!("worst error without gate: {:.3} rad", run(false));
    println!("worst error with gate:    {:.3} rad", run(true));
}
\end{verbatim}
prints \verb!worst error without gate: 0.558 rad! and \verb!worst error with gate: 0.000 rad!. Without the check the push is read as a tilt of $\operatorname{atan2}(8.98,\,9.76)=0.74$\,rad and the estimate is pulled to it; with the check the gyroscope (here exact) carries the filter through the push. Weaknesses: (1) a push or a sequence of steps that lasts long keeps the gate closed, so the gyroscope drifts uncorrected (while walking $|a|$ is almost never close to $g$; the threshold must be tuned on real data, or the weight of the accelerometer reduced smoothly instead of switching it off); (2) accelerations below the threshold pass: a sideways acceleration of $0.4$\,m/s$^2$ (the length changes by only $0.008$\,m/s$^2$) still tilts the computed angle by $\arctan(0.4/9.81)=2.3^{\circ}$, since the length of a vector grows only quadratically with a perpendicular addition, while the angle changes linearly: the length is a weak test for small sideways accelerations.
\end{enumerate}
