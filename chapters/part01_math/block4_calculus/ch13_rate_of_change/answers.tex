\begin{enumerate}[label=\textbf{\ansnum.\arabic*}]
  \item $\frac{17-5}{4}=3$. For a straight line the average rate of change equals the slope everywhere.
  \item $s(4)=20$, $s(0)=0$, so $5$ m/s.
  \item $\frac{(2+h)^2-4}{h}=4+h\to 4$.
  \item $\frac{s(3)-s(1)}{2}=\frac{18-2}{2}=8$ m/s.
  \item[\textbf{1.5}] $h=1$: $7$; $h=0.1$: $6.1$; $h=0.01$: $6.01$; $h=0.001$: $6.001$. The limit is $6$.
\end{enumerate}
