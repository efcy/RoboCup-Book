import numpy as np
def sample(logits, T=1.0, top_k=None, top_p=None, rng=None):
    rng = rng or np.random.default_rng()
    z = np.asarray(logits, float) / T
    p = np.exp(z - z.max()); p /= p.sum()
    order = np.argsort(p)[::-1]
    keep = np.ones(len(p), bool)
    if top_k is not None:
        keep[order[top_k:]] = False
    if top_p is not None:
        cum = np.cumsum(p[order])
        outside = np.concatenate([[False], cum[:-1] >= top_p])
        keep[order[outside]] = False
    q = np.where(keep, p, 0.0); q /= q.sum()
    return int(rng.choice(len(q), p=q))
\end{verbatim}
