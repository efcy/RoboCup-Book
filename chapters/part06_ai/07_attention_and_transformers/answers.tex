def mha(X, Wq, Wk, Wv, Wo, h):
    n, d = X.shape; dh = d // h
    Q, K, V = X @ Wq, X @ Wk, X @ Wv
    heads, weights = [], []
    for i in range(h):
        s = slice(i * dh, (i + 1) * dh)
        A = softmax(Q[:, s] @ K[:, s].T / np.sqrt(dh))
        heads.append(A @ V[:, s]); weights.append(A)
    return np.concatenate(heads, axis=1) @ Wo, weights
\end{verbatim}
