import cv2, numpy as np
from my_robot_vision.ball_logic import find_ball

def test_finds_orange_circle():
    img = np.zeros((120, 160, 3), np.uint8)
    orange = (0, 128, 255)       # BGR; hue 15 in OpenCV's 0-179 scale
    cv2.circle(img, (100, 60), 10, orange, -1)
    u, v, d = find_ball(img)
    assert abs(u - 100) < 1 and abs(v - 60) < 1
    assert 18 <= d <= 22

def test_empty_image():
    assert find_ball(np.zeros((120, 160, 3), np.uint8)) is None
\end{verbatim}
